class _GuiaGoAppState extends ConsumerState<GuiaGoApp> {
  StreamSubscription<String>? _tapSubscription;

  @override
  void initState() {
    super.initState();
    final AlarmNotifier notifications = ref.read(notificationServiceProvider);
    _tapSubscription = notifications.onNotificationTapped.listen(_openAlarm);
    unawaited(_checkLaunchPayload(notifications));

  void dispose() {
    _tapSubscription?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final GoRouter router = ref.watch(appRouterProvider);

    return MaterialApp.router(
      title: 'GuiaGo',
      debugShowCheckedModeBanner: false,
      theme: AppTheme.dark(),
      routerConfig: router,
    );

class AlarmRingUiState {
  const AlarmRingUiState({

    required this.occurrence,
    required this.isResolved,
  });

  final Task task;
  final TaskOccurrence occurrence;
  final bool isResolved;

  AlarmRingUiState copyWith({TaskOccurrence? occurrence, bool? isResolved}) {

/// Drives the alarm-ringing experience for one occurrence (master
/// spec §4/§24): marks it TRIGGERED, speaks the task aloud via TTS,
/// arms an automatic MISS once the tolerance window closes, and
/// forwards the user's Concluída/Adiar/Não-vou choice to [TaskEngine].
///
/// The occurrence ID is a "family" argument. Riverpod 3.0 removed
/// `FamilyAsyncNotifier`/`build(arg)` — family notifiers now receive
/// their argument via the constructor instead, and read it back from
/// a field inside `build()`.
class AlarmRingController extends AsyncNotifier<AlarmRingUiState> {
  AlarmRingController(this.occurrenceId);

  final String occurrenceId;

  Timer? _missTimer;

  @override
  Future<AlarmRingUiState> build() async {
    ref.onDispose(() => _missTimer?.cancel());

    final TaskEngine taskEngine = ref.watch(taskEngineWithSchedulerProvider);
    final taskRepository = ref.watch(taskRepositoryProvider);

    final Result<TaskOccurrence> triggerResult =
        await taskEngine.triggerOccurrence(occurrenceId);
    TaskOccurrence? occurrence = triggerResult.valueOrNull;

    // Re-opening the screen on an already-TRIGGERED occurrence is not
    // an error — just load it as-is instead of re-triggering.
    occurrence ??= (await ref
            .read(taskOccurrenceRepositoryProvider)
            .getOccurrence(occurrenceId))
        .valueOrNull;
    if (occurrence == null) {
      throw StateError('Occurrence $occurrenceId not found');
    }

    final Task? task =
        (await taskRepository.getTask(occurrence.taskId)).valueOrNull;
    if (task == null) {
      throw StateError('Task ${occurrence.taskId} not found');
    }

    await ref.read(ttsServiceProvider).speak(task.title);
    _armMissTimer(occurrence);

    return AlarmRingUiState(
      task: task,
      occurrence: occurrence,
      isResolved: false,
    );
  }

  void _armMissTimer(TaskOccurrence occurrence) {
    final Duration remaining = ToleranceChecker.remaining(
      triggeredAt: occurrence.triggeredAt ?? DateTime.now(),
      toleranceMinutes: occurrence.toleranceMinutes,
      now: DateTime.now(),
    );
    if (remaining.isNegative) {
      unawaited(_resolve(() => _engine.markMissed(occurrence.id)));
      return;
    }
    _missTimer = Timer(remaining, () {
      unawaited(_resolve(() => _engine.markMissed(occurrence.id)));
    });
  }

  Future<void> complete() {
    _missTimer?.cancel();
    final String currentOccurrenceId = _currentState!.occurrence.id;
    return _resolve(() => _engine.completeOccurrence(currentOccurrenceId));
  }

  Future<void> decline(DeclineReason reason) {
    _missTimer?.cancel();
    final String currentOccurrenceId = _currentState!.occurrence.id;
    return _resolve(
      () => _engine.declineOccurrence(
        currentOccurrenceId,
        reason: reason.label,
      ),
    );
  }

  Future<void> snooze() {
    _missTimer?.cancel();
    final String currentOccurrenceId = _currentState!.occurrence.id;
    return _resolve(() => _engine.snoozeOccurrence(currentOccurrenceId));
  }

  TaskEngine get _engine => ref.read(taskEngineWithSchedulerProvider);

  /// `AsyncValue.valueOrNull` was removed in Riverpod 3.0 — this is
  /// the pattern-matching replacement used throughout this project
  /// (see `OnboardingController` from Fase 1) since it doesn't
  /// depend on exactly which convenience getter the installed
  /// version does or doesn't have.
  AlarmRingUiState? get _currentState => switch (state) {
        AsyncData<AlarmRingUiState>(:final value) => value,
        _ => null,
      };

  Future<void> _resolve(
    Future<Result<TaskOccurrence>> Function() action,
  ) async {
    final Result<TaskOccurrence> result = await action();
    final AlarmRingUiState? current = _currentState;
    if (current == null) return;
    if (result.isSuccess) {
      state = AsyncData<AlarmRingUiState>(current.copyWith(isResolved: true));
    }
  }
}

/// The Scheduler Engine (master spec §13): schedule, cancel,
/// reschedule, rescheduleAll (the basis for Fase 7's recovery).
///
/// Implements [AlarmScheduler] so `TaskEngine` can call it without
/// knowing anything about notifications, timezones, or Drift.
class SchedulerEngine implements AlarmScheduler {
  SchedulerEngine({
    required AlarmNotifier notificationService,
    required AlarmRepository alarmRepository,
    required TaskOccurrenceRepository occurrenceRepository,
    required TaskRepository taskRepository,
  })  : _notifications = notificationService,
        _alarms = alarmRepository,
        _occurrences = occurrenceRepository,
        _tasks = taskRepository;

  final AlarmNotifier _notifications;
  final AlarmRepository _alarms;
  final TaskOccurrenceRepository _occurrences;
  final TaskRepository _tasks;

  @override
  Future<Result<void>> scheduleOccurrence(TaskOccurrence occurrence) async {
    final Result<Task?> taskResult = await _tasks.getTask(occurrence.taskId);
    final Task? task = taskResult.valueOrNull;
    final String title = task?.title ?? 'GuiaGo';

    final int nativeId = AlarmIdGenerator.fromOccurrenceId(occurrence.id);
    final Result<void> scheduleResult = await _notifications.scheduleAt(
      id: nativeId,
      title: title,
      body: 'Está agendado para agora.',
      scheduledAt: occurrence.scheduledAt,
      payload: occurrence.id,
    );
    if (scheduleResult case Fail<void>(:final failure)) {
      return Fail<void>(failure);
    }

    final AlarmRecord record = AlarmRecord(
      occurrenceId: occurrence.id,
      taskId: occurrence.taskId,
      nativeAlarmId: nativeId,
      scheduledAt: occurrence.scheduledAt,
      createdAt: DateTime.now(),
    );
    final Result<AlarmRecord> saveResult =
        await _alarms.saveAlarmRecord(record);
    if (saveResult case Fail<AlarmRecord>(:final failure)) {
      return Fail<void>(failure);
    }
    return const Success<void>(null);
  }

  @override
  Future<Result<void>> cancelOccurrence(String occurrenceId) async {
    final Result<AlarmRecord?> recordResult =
        await _alarms.getAlarmRecord(occurrenceId);
    final AlarmRecord? record = recordResult.valueOrNull;
    if (record == null) return const Success<void>(null);

    final Result<void> cancelResult =
        await _notifications.cancel(record.nativeAlarmId);
    if (cancelResult case Fail<void>(:final failure)) {
      return Fail<void>(failure);
    }
    return _alarms.deleteAlarmRecord(occurrenceId);
  }

  /// Cancels and re-schedules a single occurrence — used after a
  /// snooze changes its `scheduledAt`.
  Future<Result<void>> reschedule(TaskOccurrence occurrence) async {
    await cancelOccurrence(occurrence.id);
    return scheduleOccurrence(occurrence);
  }

  /// Re-schedules every currently-live occurrence. This is the
  /// engine-level "recovery" master spec §13 asks for — Fase 7 will
  /// call this on boot; for now it's also usable manually (e.g.
  /// right after the user grants notification permission).
  Future<Result<void>> rescheduleAll() async {
    final Result<List<TaskOccurrence>> occurrencesResult =
        await _occurrences.getAllActiveOccurrences();
    if (occurrencesResult case Fail<List<TaskOccurrence>>(:final failure)) {
      return Fail<void>(failure);
    }
    for (final TaskOccurrence occurrence in occurrencesResult.valueOrNull!) {
      await cancelOccurrence(occurrence.id);
      await scheduleOccurrence(occurrence);
    }
    return const Success<void>(null);
  }
}

/// Thin wrapper around `flutter_local_notifications` — the only
/// place in the app that talks to the native notification/alarm
/// APIs (master spec §3). Timezone handling follows master spec §14
/// — see ADR-007.
///
/// Notification-tap handling is deliberately the simple case only —
/// tap opens the app and navigates to the alarm screen, not a
/// full-screen intent. See ADR-008.
class NotificationService implements AlarmNotifier {
  NotificationService({FlutterLocalNotificationsPlugin? plugin})
      : _plugin = plugin ?? FlutterLocalNotificationsPlugin();

  final FlutterLocalNotificationsPlugin _plugin;
  bool _initialized = false;

  final StreamController<String> _tapController =
      StreamController<String>.broadcast();

  static const String _channelId = 'guiago_alarms';
  static const String _channelName = 'GuiaGo — Lembretes';
  static const String _channelDescription =
      'Lembretes de tarefas agendados pelo GuiaGo';

  @override
  Stream<String> get onNotificationTapped => _tapController.stream;

  Future<Result<void>> initialize() async {
    if (_initialized) return const Success<void>(null);
    try {
      tz_data.initializeTimeZones();
      final TimezoneInfo deviceTimezone =
          await FlutterTimezone.getLocalTimezone();
      tz.setLocalLocation(tz.getLocation(deviceTimezone.identifier));

      const AndroidInitializationSettings androidSettings =
          AndroidInitializationSettings('@mipmap/ic_launcher');
      const DarwinInitializationSettings iosSettings =
          DarwinInitializationSettings();
      const InitializationSettings settings = InitializationSettings(
        android: androidSettings,
        iOS: iosSettings,
      );
      await _plugin.initialize(
        settings: settings,
        onDidReceiveNotificationResponse: _handleTap,
      );

      // The Android notification channel (id/name/description
      // above) is created automatically the first time a
      // notification using it is shown, from the
      // AndroidNotificationDetails passed in scheduleAt() below —
      // no separate explicit creation step needed.

      _initialized = true;
      return const Success<void>(null);
    } catch (e) {
      return Fail<void>(
        PlatformFailure('Failed to initialize notifications', cause: e),
      );
    }
  }

  void _handleTap(NotificationResponse response) {
    final String? payload = response.payload;
    if (payload != null && payload.isNotEmpty) {
      _tapController.add(payload);
    }
  }

  @override
  Future<String?> checkLaunchPayload() async {
    await _ensureInitialized();
    final NotificationAppLaunchDetails? details =
        await _plugin.getNotificationAppLaunchDetails();
    if (details == null || !details.didNotificationLaunchApp) return null;
    return details.notificationResponse?.payload;
  }

  /// Requests notification + exact-alarm permissions. Call this from
  /// onboarding or first task creation — never silently.
  Future<Result<bool>> requestPermissions() async {
    try {
      final PermissionStatus notificationStatus =
          await Permission.notification.request();
      final PermissionStatus exactAlarmStatus =
          await Permission.scheduleExactAlarm.request();
      return Success<bool>(
        notificationStatus.isGranted && exactAlarmStatus.isGranted,
      );
    } catch (e) {
      return Fail<bool>(
        PermissionFailure(
          'Failed to request notification permissions',
          cause: e,
        ),
      );
    }
  }

  @override
  Future<Result<void>> scheduleAt({
    required int id,
    required String title,
    required String body,
    required DateTime scheduledAt,
    required String payload,
  }) async {
    try {
      await _ensureInitialized();
      final tz.TZDateTime scheduled = tz.TZDateTime.from(scheduledAt, tz.local);
      await _plugin.zonedSchedule(
        id: id,
        title: title,
        body: body,
        scheduledDate: scheduled,
        payload: payload,
        notificationDetails: const NotificationDetails(
          android: AndroidNotificationDetails(
            _channelId,
            _channelName,
            channelDescription: _channelDescription,
            importance: Importance.max,
            priority: Priority.high,
          ),
          iOS: DarwinNotificationDetails(),
        ),
        androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
      );
      return const Success<void>(null);
    } catch (e) {
      return Fail<void>(AlarmFailure('Failed to schedule alarm', cause: e));
    }
  }

  @override
  Future<Result<void>> cancel(int id) async {
    try {
      await _plugin.cancel(id: id);
      return const Success<void>(null);
    } catch (e) {
      return Fail<void>(AlarmFailure('Failed to cancel alarm', cause: e));
    }
  }

  @override
  Future<Result<void>> cancelAll() async {
    try {
      await _plugin.cancelAll();
      return const Success<void>(null);
    } catch (e) {
      return Fail<void>(AlarmFailure('Failed to cancel all alarms', cause: e));
    }
  }

  Future<void> _ensureInitialized() async {
    if (!_initialized) await initialize();
  }
}

/// The Task Engine (master spec §17). This is the ONLY place allowed
/// to change an occurrence's status — always through
/// [OccurrenceStateMachine] — and the only place that decides when
/// the Recurrence Engine should generate the next occurrence.
///
/// [alarmScheduler] is optional: when null (as in every Fase 2 test),
/// the engine works exactly as it did before Fase 3 — no scheduling
/// side effects. When provided, occurrence creation/resolution/snooze
/// also keeps native alarms in sync — best-effort, never blocking the
/// underlying operation on a scheduling failure (see ADR-006).
///
/// [outcomeRecorder] is the same kind of optional, best-effort hook,
/// added in Fase 5: when provided, every terminal resolution
/// (completed/declined/missed) is reported to it — e.g. so the
/// Performance feature can score it — without TaskEngine ever
/// importing that feature.
class TaskEngine {
  TaskEngine({
    required TaskRepository taskRepository,
    required TaskOccurrenceRepository occurrenceRepository,
    required TaskEventRepository eventRepository,
    AlarmScheduler? alarmScheduler,
    OccurrenceOutcomeRecorder? outcomeRecorder,
    Uuid? uuid,
  })  : _tasks = taskRepository,
        _occurrences = occurrenceRepository,
        _events = eventRepository,
        _alarmScheduler = alarmScheduler,
        _outcomeRecorder = outcomeRecorder,
        _uuid = uuid ?? const Uuid();

  final TaskRepository _tasks;
  final TaskOccurrenceRepository _occurrences;
  final TaskEventRepository _events;
  final AlarmScheduler? _alarmScheduler;
  final OccurrenceOutcomeRecorder? _outcomeRecorder;
  final Uuid _uuid;

  /// Creates a task and its first occurrence at [scheduledAt].
  /// Further occurrences are generated one at a time as each prior
  /// one resolves (§12) — never generated up front.
  Future<Result<Task>> createTask({
    required String title,
    String? description,
    required TaskPriority priority,
    required RecurrenceRule recurrenceRule,
    required DateTime scheduledAt,
    int? toleranceMinutes,
    int? snoozeMinutes,
  }) async {
    final String trimmedTitle = title.trim();
    if (trimmedTitle.isEmpty) {
      return const Fail<Task>(ValidationFailure('Task title is required'));
    }

    final DateTime now = DateTime.now();
    final Task task = Task(
      id: _uuid.v4(),
      title: trimmedTitle,
      description: description,
      priority: priority,
      recurrenceRule: recurrenceRule,
      defaultToleranceMinutes: toleranceMinutes ?? 15,
      defaultSnoozeMinutes: snoozeMinutes ?? 10,
      createdAt: now,
      updatedAt: now,
      isArchived: false,
    );

    final Result<Task> saveResult = await _tasks.saveTask(task);
    if (saveResult case Fail<Task>(:final failure)) {
      return Fail<Task>(failure);
    }

    final TaskOccurrence occurrence = TaskOccurrence(
      id: _uuid.v4(),
      taskId: task.id,
      scheduledAt: scheduledAt,
      status: OccurrenceStatus.scheduled,
      snoozeCount: 0,
      toleranceMinutes: task.defaultToleranceMinutes,
    );
    final Result<TaskOccurrence> occurrenceResult =
        await _occurrences.saveOccurrence(occurrence);
    if (occurrenceResult case Fail<TaskOccurrence>(:final failure)) {
      return Fail<Task>(failure);
    }

    await _alarmScheduler?.scheduleOccurrence(occurrence);

    return Success<Task>(task);
  }

  Future<Result<Task>> updateTask(Task task) {
    final Task updated = task.copyWith(updatedAt: DateTime.now());
    return _tasks.saveTask(updated);
  }

  Future<Result<void>> archiveTask(String taskId) => _tasks.archiveTask(taskId);

  Future<Result<void>> deleteTask(String taskId) => _tasks.deleteTask(taskId);

  /// Marks the occurrence triggered — the alarm has fired and the
  /// user is looking at it. Called by `AlarmRingController` when the
  /// alarm screen opens (Fase 4).
  Future<Result<TaskOccurrence>> triggerOccurrence(String occurrenceId) async {
    final Result<TaskOccurrence?> fetchResult =
        await _occurrences.getOccurrence(occurrenceId);
    if (fetchResult case Fail<TaskOccurrence?>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }
    final TaskOccurrence? current = fetchResult.valueOrNull;
    if (current == null) {
      return Fail<TaskOccurrence>(
        ValidationFailure('Occurrence $occurrenceId not found'),
      );
    }

    final Result<OccurrenceStatus> transition =
        OccurrenceStateMachine.transition(
      current.status,
      OccurrenceStatus.triggered,
    );
    if (transition case Fail<OccurrenceStatus>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }

    final TaskOccurrence triggered = current.copyWith(
      status: OccurrenceStatus.triggered,
      triggeredAt: DateTime.now(),
    );
    final Result<TaskOccurrence> saveResult =
        await _occurrences.saveOccurrence(triggered);
    if (saveResult case Fail<TaskOccurrence>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }

    await _recordEvent(current, OccurrenceStatus.triggered);

    return Success<TaskOccurrence>(triggered);
  }

  /// Marks the occurrence completed and, if the task recurs,
  /// generates and schedules the next occurrence.
  Future<Result<TaskOccurrence>> completeOccurrence(
    String occurrenceId, {
    Duration? responseTime,
  }) {
    return _resolveOccurrence(
      occurrenceId,
      OccurrenceStatus.completed,
      applyExtras: (TaskOccurrence occ) => occ.copyWith(
        completedAt: DateTime.now(),
        responseTime: responseTime,
      ),
    );
  }

  /// Marks the occurrence declined (optionally with a reason) and,
  /// if the task recurs, generates and schedules the next occurrence.
  Future<Result<TaskOccurrence>> declineOccurrence(
    String occurrenceId, {
    String? reason,
  }) {
    return _resolveOccurrence(
      occurrenceId,
      OccurrenceStatus.declined,
      applyExtras: (TaskOccurrence occ) => occ.copyWith(declineReason: reason),
    );
  }

  /// Marks the occurrence missed and, if the task recurs, generates
  /// and schedules the next occurrence.
  Future<Result<TaskOccurrence>> markMissed(String occurrenceId) {
    return _resolveOccurrence(occurrenceId, OccurrenceStatus.missed);
  }

  /// Snoozes the occurrence: increments `snoozeCount`, pushes
  /// `scheduledAt` forward by [snoozeMinutes], and re-arms the alarm
  /// for the new time. Per §11, SNOOZED must return to SCHEDULED — it
  /// never resolves the occurrence, so no next occurrence is
  /// generated here.
  Future<Result<TaskOccurrence>> snoozeOccurrence(
    String occurrenceId, {
    int snoozeMinutes = 10,
  }) async {
    final Result<TaskOccurrence?> fetchResult =
        await _occurrences.getOccurrence(occurrenceId);
    if (fetchResult case Fail<TaskOccurrence?>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }
    final TaskOccurrence? current = fetchResult.valueOrNull;
    if (current == null) {
      return Fail<TaskOccurrence>(
        ValidationFailure('Occurrence $occurrenceId not found'),
      );
    }

    final Result<OccurrenceStatus> snoozedTransition =
        OccurrenceStateMachine.transition(
      current.status,
      OccurrenceStatus.snoozed,
    );
    if (snoozedTransition case Fail<OccurrenceStatus>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }
    await _recordEvent(current, OccurrenceStatus.snoozed);

    final Result<OccurrenceStatus> rescheduleTransition =
        OccurrenceStateMachine.transition(
      OccurrenceStatus.snoozed,
      OccurrenceStatus.scheduled,
    );
    if (rescheduleTransition case Fail<OccurrenceStatus>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }

    final TaskOccurrence rescheduled = current.copyWith(
      status: OccurrenceStatus.scheduled,
      snoozeCount: current.snoozeCount + 1,
      scheduledAt: DateTime.now().add(Duration(minutes: snoozeMinutes)),
    );
    final Result<TaskOccurrence> saveResult =
        await _occurrences.saveOccurrence(rescheduled);
    if (saveResult case Fail<TaskOccurrence>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }
    await _recordEvent(
      current.copyWith(status: OccurrenceStatus.snoozed),
      OccurrenceStatus.scheduled,
    );

    await _alarmScheduler?.cancelOccurrence(occurrenceId);
    await _alarmScheduler?.scheduleOccurrence(rescheduled);

    return Success<TaskOccurrence>(rescheduled);
  }

  /// Shared implementation for the three terminal transitions
  /// (completed/declined/missed): validate via the state machine,
  /// persist, log the [TaskEvent], cancel the resolved occurrence's
  /// alarm, then generate + schedule the task's next occurrence if
  /// it recurs.
  Future<Result<TaskOccurrence>> _resolveOccurrence(
    String occurrenceId,
    OccurrenceStatus toStatus, {
    TaskOccurrence Function(TaskOccurrence occurrence)? applyExtras,
  }) async {
    final Result<TaskOccurrence?> fetchResult =
        await _occurrences.getOccurrence(occurrenceId);
    if (fetchResult case Fail<TaskOccurrence?>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }
    final TaskOccurrence? current = fetchResult.valueOrNull;
    if (current == null) {
      return Fail<TaskOccurrence>(
        ValidationFailure('Occurrence $occurrenceId not found'),
      );
    }

    final Result<OccurrenceStatus> transition =
        OccurrenceStateMachine.transition(current.status, toStatus);
    if (transition case Fail<OccurrenceStatus>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }

    TaskOccurrence resolved = current.copyWith(status: toStatus);
    if (applyExtras != null) {
      resolved = applyExtras(resolved);
    }

    final Result<TaskOccurrence> saveResult =
        await _occurrences.saveOccurrence(resolved);
    if (saveResult case Fail<TaskOccurrence>(:final failure)) {
      return Fail<TaskOccurrence>(failure);
    }

    await _recordEvent(current, toStatus);
    await _alarmScheduler?.cancelOccurrence(occurrenceId);

    final TaskOccurrence? next =
        await _generateNextOccurrenceIfRecurring(resolved);
    if (next != null) {
      await _alarmScheduler?.scheduleOccurrence(next);
    }

    await _outcomeRecorder?.onOccurrenceResolved(resolved);

    return Success<TaskOccurrence>(resolved);
  }

  Future<TaskOccurrence?> _generateNextOccurrenceIfRecurring(
    TaskOccurrence resolvedOccurrence,
  ) async {
    final Result<Task?> taskResult =
        await _tasks.getTask(resolvedOccurrence.taskId);
    final Task? task = taskResult.valueOrNull;
    if (task == null || task.isArchived) return null;

    final DateTime? next = RecurrenceEngine.nextOccurrenceAfter(
      recurrenceRule: task.recurrenceRule,
      anchor: resolvedOccurrence.scheduledAt,
      after: resolvedOccurrence.scheduledAt,
    );
    if (next == null) return null;

    final TaskOccurrence nextOccurrence = TaskOccurrence(
      id: _uuid.v4(),
      taskId: task.id,
      scheduledAt: next,
      status: OccurrenceStatus.scheduled,
      snoozeCount: 0,
      toleranceMinutes: task.defaultToleranceMinutes,
    );
    final Result<TaskOccurrence> saveResult =
        await _occurrences.saveOccurrence(nextOccurrence);
    return saveResult.valueOrNull;
  }

  Future<void> _recordEvent(
    TaskOccurrence before,
    OccurrenceStatus toStatus,
  ) {
    return _events.recordEvent(
      TaskEvent(
        id: _uuid.v4(),
        taskId: before.taskId,
        occurrenceId: before.id,
        fromStatus: before.status,
        toStatus: toStatus,
        occurredAt: DateTime.now(),
      ),
    );
  }
}
