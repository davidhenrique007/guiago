import 'package:flutter/material.dart';

import '../../app/theme/app_colors.dart';

/// Bar chart of one value per day, drawn with [CustomPainter] — no
/// charting dependency. The master spec (§4) doesn't list one, and
/// this project's Riverpod/Freezed versions are recent enough that
/// adding a charting package without verifying compatibility first
/// would risk exactly the kind of breakage already seen in Fase 5.
///
/// Deliberately feature-agnostic: takes plain doubles, not a
/// dashboard-specific type, so design_system/ never has to import a
/// feature.
class AppTrendChart extends StatelessWidget {
  const AppTrendChart({required this.values, this.height = 120, super.key});

  /// One value per bar, oldest first, each normally 0.0–1.0.
  /// A negative value means "no data that day" and draws as a
  /// small empty tick instead of a bar.
  final List<double> values;
  final double height;

  @override
  Widget build(BuildContext context) {
    return SizedBox(
      height: height,
      width: double.infinity,
      child: CustomPaint(painter: _TrendChartPainter(values: values)),
    );
  }
}

class _TrendChartPainter extends CustomPainter {
  _TrendChartPainter({required this.values});

  final List<double> values;

  @override
  void paint(Canvas canvas, Size size) {
    if (values.isEmpty) return;

    final double barWidth = size.width / values.length;
    final double gap = barWidth * 0.28;
    final Paint filledPaint = Paint()..color = AppColors.primary;
    final Paint emptyPaint = Paint()..color = AppColors.surfaceMuted;

    for (int i = 0; i < values.length; i++) {
      final double value = values[i];
      final bool hasData = value >= 0;
      final double clamped = hasData ? value.clamp(0.0, 1.0) : 0.0;
      final double barHeight =
          hasData ? (size.height * clamped).clamp(4.0, size.height) : 4.0;

      final double left = i * barWidth + gap / 2;
      final double right = (i + 1) * barWidth - gap / 2;
      final double top = size.height - barHeight;

      final RRect rect = RRect.fromRectAndRadius(
        Rect.fromLTRB(left, top, right, size.height),
        const Radius.circular(3),
      );
      canvas.drawRRect(rect, hasData ? filledPaint : emptyPaint);
    }
  }

  @override
  bool shouldRepaint(covariant _TrendChartPainter oldDelegate) {
    return oldDelegate.values != values;
  }
}

  const HomeGreeting({required this.userName, required this.today, super.key});

  final String userName;
  final DateTime today;

  static const List<String> _weekdays = <String>[
    'segunda-feira',
    'terça-feira',
    'quarta-feira',
    'quinta-feira',
    'sexta-feira',
    'sábado',
    'domingo',
  ];

  static const List<String> _months = <String>[
    'janeiro',
    'fevereiro',
    'março',
    'abril',
    'maio',
    'junho',
    'julho',
    'agosto',
    'setembro',
    'outubro',
    'novembro',
    'dezembro',
  ];

  String get _period {
    final int hour = DateTime.now().hour;
    if (hour < 12) return 'Bom dia';
    if (hour < 19) return 'Boa tarde';
    return 'Boa noite';
  }

  String get _dateLabel {
    final String weekday = _weekdays[today.weekday - 1];
    final String month = _months[today.month - 1];
    return '${_capitalize(weekday)}, ${today.day} de $month';
  }

  static String _capitalize(String value) =>
      '${value[0].toUpperCase()}${value.substring(1)}';

  @override
  Widget build(BuildContext context) {
    final String suffix = userName.trim().isEmpty ? '' : ', $userName';
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: <Widget>[
        Text('$_period$suffix', style: AppTypography.displayLarge),
        const SizedBox(height: AppSpacing.xs),
        Text(_dateLabel, style: AppTypography.bodyMedium),
      ],
    );
  }
}

import '../domain/level.dart';
import '../domain/level_policy.dart';
import '../domain/performance_record.dart';

class LevelEngine {
  const LevelEngine({this.policy = LevelPolicy.defaultPolicy});

  final LevelPolicy policy;

  /// [id] é gerado pelo chamador (uuid), consistente com o padrão
  /// já usado nas fases anteriores. [recentPerformance] deve cobrir
  /// uma janela curta (ex.: últimos 7 ou 14 dias) — é o que impede
  /// um score alto e antigo de manter alguém em ELITE se o
  /// desempenho recente caiu.
  LevelStatus evaluate({
    required String id,
    required int totalScore,
    required LevelTier previousTier,
    required PerformanceRecord recentPerformance,
    required DateTime evaluatedAt,
  }) {
    var tier = policy.tierForScore(totalScore);

    final belowFloor = !recentPerformance.isEmpty &&
        recentPerformance.completionRate < policy.recentCompletionRateFloor;

    if (belowFloor && tier.rank > LevelTier.beginner.rank) {
      tier = LevelTier.values[tier.rank - 1];
    }

    return LevelStatus(
      id: id,
      tier: tier,
      previousTier: previousTier,
      scoreAtEvaluation: totalScore,
      evaluatedAt: evaluatedAt,
    );
  }
}

// GuiaGo — Fase 5: Performance Engine + Gamification
// PerformanceEngine (secção 18 da master spec)
//
// Função pura: recebe outcomes de um período, devolve um
// PerformanceRecord. Sem I/O, sem SQLite, sem Flutter — por isso
// não usa Result<T> aqui (não há falha de infraestrutura
// possível; lista vazia é um caso válido, não um erro).

import '../domain/occurrence_outcome.dart';
import '../domain/performance_record.dart';

class PerformanceEngine {
  const PerformanceEngine();

  PerformanceRecord calculate({
    required String id,
    required DateTime periodStart,
    required DateTime periodEnd,
    required List<OccurrenceOutcome> outcomes,
    required DateTime calculatedAt,
  }) {
    assert(
      !periodEnd.isBefore(periodStart),
      'periodEnd não pode ser anterior a periodStart',
    );

    final total = outcomes.length;

    if (total == 0) {
      return PerformanceRecord(
        id: id,
        periodStart: periodStart,
        periodEnd: periodEnd,
        totalOccurrences: 0,
        completedCount: 0,
        completedAfterSnoozeCount: 0,
        missedCount: 0,
        declinedCount: 0,
        snoozedCount: 0,
        completionRate: 0,
        missRate: 0,
        snoozeRate: 0,
        averageResponseTime: null,
        calculatedAt: calculatedAt,
      );
    }

    final completed =
        outcomes.where((o) => o.status == OutcomeStatus.completed).toList();
    final completedAfterSnooze =
        completed.where((o) => o.snoozeCount > 0).toList();
    final missed =
        outcomes.where((o) => o.status == OutcomeStatus.missed).toList();
    final declined =
        outcomes.where((o) => o.status == OutcomeStatus.declined).toList();
    final snoozed = outcomes.where((o) => o.snoozeCount > 0).toList();

    final responseTimes =
        outcomes.map((o) => o.responseTime).whereType<Duration>().toList();

    final averageResponseTime = responseTimes.isEmpty
        ? null
        : Duration(
            microseconds: responseTimes.fold<int>(
                  0,
                  (sum, d) => sum + d.inMicroseconds,
                ) ~/
                responseTimes.length,
          );

    return PerformanceRecord(
      id: id,
      periodStart: periodStart,
      periodEnd: periodEnd,
      totalOccurrences: total,
      completedCount: completed.length,
      completedAfterSnoozeCount: completedAfterSnooze.length,
      missedCount: missed.length,
      declinedCount: declined.length,
      snoozedCount: snoozed.length,
      completionRate: completed.length / total,
      missRate: missed.length / total,
      snoozeRate: snoozed.length / total,
      averageResponseTime: averageResponseTime,
      calculatedAt: calculatedAt,
    );
  }
}
