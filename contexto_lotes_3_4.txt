/// Thrown into the [AsyncValue] when the snapshot cannot be loaded, so
/// the UI can show the original [failure] in an error state.
class DashboardLoadException implements Exception {
  const DashboardLoadException(this.failure);

  final Failure failure;

  @override
  String toString() => 'DashboardLoadException: ${failure.message}';
}

/// One controller per [DashboardPeriod] (Riverpod family, same shape
/// as `AlarmRingController`: the argument arrives via the constructor).
class DashboardController extends AsyncNotifier<DashboardSnapshot> {
  DashboardController(this.period);

  final DashboardPeriod period;

  @override
  Future<DashboardSnapshot> build() async {
    final Result<DashboardSnapshot> result =
        await ref.read(dashboardServiceProvider).load(period);
    return switch (result) {
      Success<DashboardSnapshot>(:final value) => value,
      Fail<DashboardSnapshot>(:final failure) =>
        throw DashboardLoadException(failure),
    };
  }

  /// Re-reads everything. The screen calls this when it opens, since
  /// occurrences may have been resolved since the last load.
  void refresh() => ref.invalidateSelf();
}

/// Read-only assembler of a [DashboardSnapshot].
///
/// Period metrics and the trend are computed in memory from the
/// occurrences of the period; Score, Level and Streak are read from
/// what `PerformanceController` already persisted. Opening the
/// dashboard therefore never writes to the database.
class DashboardService {
  DashboardService({
    required TaskOccurrenceRepository occurrenceRepository,
    required OccurrenceOutcomeMapper mapper,
    required PerformanceEngine performanceEngine,
    required TrendEngine trendEngine,
    required ScoreRepository scoreRepository,
    required LevelRepository levelRepository,
    required StreakRepository streakRepository,
  })  : _occurrences = occurrenceRepository,
        _mapper = mapper,
        _performanceEngine = performanceEngine,
        _trendEngine = trendEngine,
        _scores = scoreRepository,
        _levels = levelRepository,
        _streaks = streakRepository;

  final TaskOccurrenceRepository _occurrences;
  final OccurrenceOutcomeMapper _mapper;
  final PerformanceEngine _performanceEngine;
  final TrendEngine _trendEngine;
  final ScoreRepository _scores;
  final LevelRepository _levels;
  final StreakRepository _streaks;

  /// [now] exists only so tests can pin the clock.
  Future<Result<DashboardSnapshot>> load(
    DashboardPeriod period, {
    DateTime? now,
  }) async {
    final DateTime clock = now ?? DateTime.now();
    final DashboardRange range = period.rangeAt(clock);

    final occurrencesResult =
        await _occurrences.getOccurrencesInRange(range.start, range.end);
    if (occurrencesResult.isFailure) {
      return Fail<DashboardSnapshot>(occurrencesResult.failureOrNull!);
    }

    final scoreResult = await _scores.getScore();
    if (scoreResult.isFailure) {
      return Fail<DashboardSnapshot>(scoreResult.failureOrNull!);
    }
    final levelResult = await _levels.getLatest();
    if (levelResult.isFailure) {
      return Fail<DashboardSnapshot>(levelResult.failureOrNull!);
    }
    final streakResult = await _streaks.getStatus();
    if (streakResult.isFailure) {
      return Fail<DashboardSnapshot>(streakResult.failureOrNull!);
    }

    final Map<DateTime, List<OccurrenceOutcome>> byDay =
        <DateTime, List<OccurrenceOutcome>>{};
    for (final TaskOccurrence occurrence in occurrencesResult.valueOrNull!) {
      final OccurrenceOutcome? outcome = _mapper.toOutcome(occurrence);
      if (outcome == null) continue; // not finished, or cancelled
      final DateTime day = DateTime(
        occurrence.scheduledAt.year,
        occurrence.scheduledAt.month,
        occurrence.scheduledAt.day,
      );
      byDay.putIfAbsent(day, () => <OccurrenceOutcome>[]).add(outcome);
    }

    return Success<DashboardSnapshot>(
      DashboardSnapshot(
        period: period,
        rangeStart: range.start,
        rangeEnd: range.end,
        performance: _performanceEngine.calculate(
          id: 'dashboard-${period.name}',
          periodStart: range.start,
          periodEnd: range.end,
          outcomes: byDay.values.expand((l) => l).toList(),
          calculatedAt: clock,
        ),
        trend: _trendEngine.calculate(
          rangeStart: range.start,
          rangeEnd: range.end,
          outcomesByDay: byDay,
        ),
        score: scoreResult.valueOrNull!,
        level: levelResult.valueOrNull,
        streak: streakResult.valueOrNull,
      ),
    );
  }
}

/// Pure: turns finished occurrences, grouped by local day, into one
/// [TrendPoint] per day of the range (days without occurrences get a
/// zeroed point, so the chart never has holes).
class TrendEngine {
  const TrendEngine();

  /// [outcomesByDay] keys must be local dates normalized to midnight.
  List<TrendPoint> calculate({
    required DateTime rangeStart,
    required DateTime rangeEnd,
    required Map<DateTime, List<OccurrenceOutcome>> outcomesByDay,
  }) {
    final List<TrendPoint> points = <TrendPoint>[];
    final DateTime last = DateTime(rangeEnd.year, rangeEnd.month, rangeEnd.day);

    for (var day = DateTime(rangeStart.year, rangeStart.month, rangeStart.day);
        !day.isAfter(last);
        day = DateTime(day.year, day.month, day.day + 1)) {
      final List<OccurrenceOutcome> outcomes =
          outcomesByDay[day] ?? const <OccurrenceOutcome>[];
      points.add(
        TrendPoint(
          day: day,
          total: outcomes.length,
          completed:
              outcomes.where((o) => o.status == OutcomeStatus.completed).length,
        ),
      );
    }
    return points;
  }
}

/// Rolling window shown by the dashboard: today, the last 7 days or
/// the last 30 days (ending today). Rolling windows avoid depending on
/// which weekday the week starts on.
enum DashboardPeriod { today, week, month }

/// Inclusive local-time range covered by a [DashboardPeriod].
typedef DashboardRange = ({DateTime start, DateTime end});

extension DashboardPeriodX on DashboardPeriod {
  int get days => switch (this) {
        DashboardPeriod.today => 1,
        DashboardPeriod.week => 7,
        DashboardPeriod.month => 30,
      };

  /// The range that ends at the last instant of [now]'s local day and
  /// starts at midnight, [days] - 1 days earlier. Uses calendar
  /// arithmetic (not `Duration`) so daylight-saving changes cannot
  /// shift the boundaries.
  DashboardRange rangeAt(DateTime now) {
    final DateTime start = DateTime(now.year, now.month, now.day - (days - 1));
    final DateTime end = DateTime(now.year, now.month, now.day + 1)
        .subtract(const Duration(microseconds: 1));
    return (start: start, end: end);
  }
}

/// Read-only assembler of a [HomeSnapshot] — same shape as
/// DashboardService (Fase 6B): never writes, only reads and joins
/// what other features already persisted.
class HomeService {
  HomeService({
    required TaskOccurrenceRepository occurrenceRepository,
    required TaskRepository taskRepository,
    required SettingsRepository settingsRepository,
  })  : _occurrences = occurrenceRepository,
        _tasks = taskRepository,
        _settings = settingsRepository;

  final TaskOccurrenceRepository _occurrences;
  final TaskRepository _tasks;
  final SettingsRepository _settings;

  /// [now] exists only so tests can pin the clock.
  Future<Result<HomeSnapshot>> load({DateTime? now}) async {
    final DateTime clock = now ?? DateTime.now();
    final DateTime dayStart = DateTime(clock.year, clock.month, clock.day);
    final DateTime dayEnd = DateTime(clock.year, clock.month, clock.day + 1)
        .subtract(const Duration(microseconds: 1));

    final settingsResult = await _settings.getSettings();
    if (settingsResult.isFailure) {
      return Fail<HomeSnapshot>(settingsResult.failureOrNull!);
    }

    final occurrencesResult = await _occurrences.getOccurrencesInRange(
      dayStart,
      dayEnd,
    );
    if (occurrencesResult.isFailure) {
      return Fail<HomeSnapshot>(occurrencesResult.failureOrNull!);
    }

    final List<TaskOccurrence> occurrences = occurrencesResult.valueOrNull!
      ..sort((a, b) => a.scheduledAt.compareTo(b.scheduledAt));

    final List<TodayTaskItem> items = <TodayTaskItem>[];
    for (final TaskOccurrence occurrence in occurrences) {
      // Cancelled occurrences (task deleted/archived) don't belong on
      // the Home list — same rule as the Fase 5 mapper.
      if (occurrence.status == OccurrenceStatus.cancelled) continue;

      final taskResult = await _tasks.getTask(occurrence.taskId);
      if (taskResult.isFailure) {
        return Fail<HomeSnapshot>(taskResult.failureOrNull!);
      }
      final Task? task = taskResult.valueOrNull;
      if (task == null) continue; // task deleted after being scheduled

      items.add(
        TodayTaskItem(
          occurrenceId: occurrence.id,
          taskId: task.id,
          title: task.title,
          priority: task.priority,
          scheduledAt: occurrence.scheduledAt,
          status: occurrence.status,
        ),
      );
    }

    TodayTaskItem? next;
    for (final TodayTaskItem item in items) {
      if (!item.isResolved) {
        next = item;
        break;
      }
    }

    return Success<HomeSnapshot>(
      HomeSnapshot(
        userName: settingsResult.valueOrNull!.userName,
        today: dayStart,
        todayTasks: items,
        nextTask: next,
      ),
    );
  }
}

/// Validates and applies [OccurrenceStatus] transitions (master spec
/// §11). Independent of UI and persistence — a pure function of
/// (current status, requested status).
///
/// `cancelled` is reachable from `scheduled` and `snoozed` (e.g. the
/// parent task gets archived/deleted) even though master spec §11's
/// transition list doesn't spell it out explicitly — otherwise
/// `cancelled` would be a status nothing could ever reach.
abstract final class OccurrenceStateMachine {
  static const Map<OccurrenceStatus, Set<OccurrenceStatus>> _validTransitions =
      <OccurrenceStatus, Set<OccurrenceStatus>>{
    OccurrenceStatus.scheduled: <OccurrenceStatus>{
      OccurrenceStatus.triggered,
      OccurrenceStatus.cancelled,
    },
    OccurrenceStatus.triggered: <OccurrenceStatus>{
      OccurrenceStatus.completed,
      OccurrenceStatus.snoozed,
      OccurrenceStatus.declined,
      OccurrenceStatus.missed,
    },
    OccurrenceStatus.snoozed: <OccurrenceStatus>{
      OccurrenceStatus.scheduled,
      OccurrenceStatus.cancelled,
    },
    OccurrenceStatus.completed: <OccurrenceStatus>{},
    OccurrenceStatus.declined: <OccurrenceStatus>{},
    OccurrenceStatus.missed: <OccurrenceStatus>{},
    OccurrenceStatus.cancelled: <OccurrenceStatus>{},
  };

  /// Whether the domain allows moving from [from] to [to] directly.
  static bool canTransition(OccurrenceStatus from, OccurrenceStatus to) =>
      _validTransitions[from]?.contains(to) ?? false;

  /// The terminal (no further transitions possible) statuses.
  static bool isTerminal(OccurrenceStatus status) =>
      _validTransitions[status]?.isEmpty ?? true;

  /// Returns [to] wrapped in [Success] if the transition is valid,
  /// or a [SchedulingFailure] if not.
  static Result<OccurrenceStatus> transition(
    OccurrenceStatus from,
    OccurrenceStatus to,
  ) {
    if (!canTransition(from, to)) {
      return Fail<OccurrenceStatus>(
        SchedulingFailure('Invalid occurrence transition: $from -> $to'),
      );
    }
    return Success<OccurrenceStatus>(to);
  }
}

/// Computes the next occurrence date/time for a recurring [Task]
/// (master spec §12).
///
/// Deliberately generates ONE occurrence at a time — call this again
/// only after the current occurrence resolves (completed / declined
/// / missed). Never pre-generates a batch, so there is no risk of
/// runaway occurrence generation.
abstract final class RecurrenceEngine {
  /// Returns the next scheduled date/time strictly after [after],
  /// preserving the time-of-day and (for `weekly`) the weekday from
  /// [anchor] — the task's original scheduled time. Returns `null`
  /// for [RecurrenceRule.none] (a one-off task has no next
  /// occurrence).
  static DateTime? nextOccurrenceAfter({
    required RecurrenceRule recurrenceRule,
    required DateTime anchor,
    required DateTime after,
  }) {
    return switch (recurrenceRule) {
      RecurrenceNone() => null,
      RecurrenceDaily() => _nextDaily(anchor, after),
      RecurrenceWeekly() =>
        _nextOnWeekdays(<int>{anchor.weekday}, anchor, after),
      RecurrenceCustomDays(:final weekdays) =>
        _nextOnWeekdays(weekdays, anchor, after),
    };
  }

  static DateTime _nextDaily(DateTime anchor, DateTime after) {
    DateTime candidate = _atTimeOfDay(after, anchor);
    if (!candidate.isAfter(after)) {
      candidate = candidate.add(const Duration(days: 1));
    }
    return candidate;
  }

  static DateTime _nextOnWeekdays(
    Set<int> weekdays,
    DateTime anchor,
    DateTime after,
  ) {
    assert(weekdays.isNotEmpty, 'weekdays must not be empty');
    DateTime candidate = _atTimeOfDay(after, anchor);
    // At most 8 days needed to guarantee we cross a full week and
    // find a matching weekday strictly after `after`.
    for (int i = 0; i < 8; i++) {
      if (weekdays.contains(candidate.weekday) && candidate.isAfter(after)) {
        return candidate;
      }
      candidate = candidate.add(const Duration(days: 1));
    }
    // Unreachable: weekdays is non-empty and every weekday recurs
    // within 7 days, so the loop above always returns first.
    throw StateError(
      'RecurrenceEngine: no matching weekday found for $weekdays',
    );
  }

  static DateTime _atTimeOfDay(DateTime date, DateTime timeSource) {
    return DateTime(
      date.year,
      date.month,
      date.day,
      timeSource.hour,
      timeSource.minute,
    );
  }
}

import '../../../core/result/result.dart';
import 'task.dart';

/// Domain-facing contract for reading/writing [Task]s. Implemented
/// by `DriftTaskRepository` in the Data layer.
abstract interface class TaskRepository {
  Future<Result<Task>> saveTask(Task task);

  Future<Result<Task?>> getTask(String id);

  /// Non-archived tasks, most recently updated first.
  Future<Result<List<Task>>> getActiveTasks();

  Future<Result<void>> archiveTask(String id);

  Future<Result<void>> deleteTask(String id);
}
